<eventstorming>
  <aggregate name="User">
    <container name="User Registration">
      <actor name="Customer" />
      <command name="Register" />
      <policy name="Is Email Valid?" negativeNext="Invalid Email" />
      <error name="Invalid Email"><note>The email address provided is not valid.</note></error>
      <event name="UserRegistered" />
    </container>
  </aggregate>
  <aggregate name="Order">
    <container name="Place Order">
      <command name="PlaceOrder" />
      <container name="Validate Order">
        <policy name="IsAddressValid" negativeNext="AddressIsInValid" />
        <policy name="IsEmailValid" negativeNext="Email is invalid" />
        <error name="AddressIsInValid"><note>The address provided is not valid.</note></error>
        <error name="Email is invalid"><note>The email address is not valid.</note></error>
      </container>
      <policy name="Is Payment Successful?" negativeNext="PaymentFailed" />
      <error name="PaymentFailed"><note>Payment failed, please try again.</note></error>
      <event name="OrderPlaced" />
    </container>
  </aggregate>
</eventstorming>
